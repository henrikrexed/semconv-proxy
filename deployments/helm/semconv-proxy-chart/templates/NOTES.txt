Quick checks:

  # Forward the API + Web UI locally
  kubectl port-forward -n {{ .Release.Namespace }} \
    svc/{{ include "semconv-proxy-chart.fullname" . }} \
    8080:{{ .Values.service.ports.api }}

  # SemConv Explorer Web UI
  #   open http://localhost:8080/

  # Health
  curl -fsS http://localhost:8080/healthz

  # Discovered attributes
  curl -fsS http://localhost:8080/api/v1/dictionary/attributes

  # Export Weaver-compatible YAML
  curl -fsS http://localhost:8080/api/v1/weaver/export > semconv.yaml
